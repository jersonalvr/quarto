% Configuración específica para el TOC
\AtBeginDocument{
  % Redefinir el comando del TOC para aplicar el estilo correcto
  \let\oldtableofcontents\tableofcontents
  \renewcommand{\tableofcontents}{%
    \pagestyle{toc}%
    \pagenumbering{roman}%
    \setcounter{page}{1}%
    \oldtableofcontents
  }
  
  % También para lista de figuras y tablas si las tienes
  \let\oldlistoffigures\listoffigures
  \renewcommand{\listoffigures}{
    \pagestyle{toc}
    \hypersetup{linkcolor=black}
    \oldlistoffigures
  }
  
  \let\oldlistoftables\listoftables
  \renewcommand{\listoftables}{
    \pagestyle{toc}
    \hypersetup{linkcolor=black}
    \oldlistoftables
    \cleardoublepage
    \pagenumbering{arabic}
    \setcounter{page}{1}
    \pagestyle{posttoc}
  }
}
